🚀 Готовая автосборка (GitHub Actions)

При каждом git push воркфлоу автоматически собирает Android APK (через Capacitor + Android SDK) и публикует мобильную веб-версию на GitHub Pages.

.github/workflows/build.yml 
name: Build Mobile Auto Quiz APK

on:
  push:
    branches: [ "**" ]
  pull_request:
    branches: [ "**" ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build-apk:
    name: Build Android APK
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: '22'

      - name: Setup Java 17 (Temurin)
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Prepare Capacitor & Build APK
        run: |
          mkdir -p www
          if [ ! -f "www/index.html" ] && [ -f "index.html" ]; then
            cp index.html www/index.html
          fi
          rm -rf node_modules package-lock.json android
          npm init -y
          npm install @capacitor/core@6.1.2 @capacitor/cli@6.1.2 @capacitor/android@6.1.2
          cat << 'EOF' > capacitor.config.json
          {
            "appId": "com.retroauto.quiz",
            "appName": "AutoQuiz",
            "webDir": "www"
          }
          EOF
          npx cap add android
          npx cap sync android
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon

      - name: Upload Android APK Artifact
        uses: actions/upload-artifact@v4
        with:
          name: auto-quiz-android-debug-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
